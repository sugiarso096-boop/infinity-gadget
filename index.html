Coba lagi."; btn.disabled = false; btn.innerHTML = ' Simpan Perubahan'; }); } function hapusBarang(event, rowNum) { if(event) event.stopPropagation(); if(confirm("Yakin ingin menghapus barang ini secara permanen?")) { let payload = { tipeForm: "hapus_barang", rowHapus: rowNum }; // Berikan feedback visual sederhana fetch(scriptURL, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify(payload) }) .then(res => res.text()) .then(msg => { alert(msg); muatSemuaBarangUntukEdit(); muatDashboard(); }) .catch(err => { alert("Gagal menghapus barang.
